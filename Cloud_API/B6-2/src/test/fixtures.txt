def calculate_total(items):
    """장바구니 아이템들의 총합을 계산한다."""
    return sum(item.price * item.quantity for item in items)


def format_currency(amount):
    """숫자를 원화 표기 문자열로 변환한다."""
    return f"{amount:,}원"


def apply_discount(total, discount_rate):
    """할인율을 적용한 최종 금액을 반환한다."""
    return round(total * (1 - discount_rate))


class ShoppingCart:
    """장바구니 아이템을 관리하는 클래스."""

    def __init__(self):
        self.items = []

    def add_item(self, item):
        self.items.append(item)

    def total_price(self):
        return calculate_total(self.items)

    def total_with_discount(self, discount_rate):
        return apply_discount(self.total_price(), discount_rate)
